\section{Paper 1 -- Audebert et al.\ (2020)}

\begin{frame}{Latar belakang, dataset, modalitas}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Masalah}
    \begin{itemize}\kecil
      \item Dokumen hasil pindai/foto tidak punya lapisan teks; pengenalan jenis dokumen masih manual.
      \item Visual saja cukup untuk kelas kasar, tidak untuk kelas halus (formulir pajak: tata letak sama, isi beda).
      \item Pertanyaan: apakah teks OCR yang berderau tetap menambah akurasi jika digabung dengan citra, dengan biaya rendah?
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Dataset}
    \begin{itemize}\kecil
      \item Tobacco3482: 3.482 dok., 10 kelas; 800 latih, sisanya uji; rerata 3 run.
      \item RVL-CDIP: 400 rb dok., 16 kelas; split 320k/40k/40k.
      \item QS-OCR: teks Tesseract kedua dataset, dirilis terbuka.
    \end{itemize}
    \textbf{Modalitas}
    \begin{itemize}\kecil
      \item Citra grayscale halaman + teks OCR dari citra yang sama.
      \item Teks adalah turunan citra; tata letak tidak dipakai.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Pipeline}
  \input{../gambar/pipeline-audebert}
  \vspace{6pt}
  \begin{itemize}\kecil
    \item Dua cabang menghasilkan vektor 128-d, lalu dikonkatenasi dan diklasifikasi MLP.
    \item Seluruh jaringan dilatih end-to-end (FastText tetap beku).
  \end{itemize}
\end{frame}

\begin{frame}{Model unimodal dan fusi}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Citra: MobileNetV2}
    \begin{itemize}\kecil
      \item Ringan; akurasi setara VGG-16, jauh lebih cepat.
      \item 384$\times$384, grayscale diduplikasi 3 kanal, bobot ImageNet.
      \item Augmentasi tidak membantu (84,5 vs.\ 83,9\,\%).
    \end{itemize}
    \textbf{Teks: FastText + CNN 1D}
    \begin{itemize}\kecil
      \item Tesseract 4.0 LSTM; $\sim$26\,\% kata tidak ada di kamus.
      \item FastText (subkata) agar kata salah eja tetap punya vektor.
      \item CNN 1D: 4 lapis, kernel 12, 512 kanal, max-pool waktu; 500 kata.
      \item MLP embedding dokumen kalah (70,8 vs.\ 73,9\,\%).
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Fusi: tingkat fitur}
    \begin{itemize}\kecil
      \item Teks 128-d $\oplus$ citra (GAP 1280 $\to$ FC 128) $\to$ MLP.
      \item Penjumlahan fitur dicoba, tetapi kalah dari citra saja.
    \end{itemize}
    \textbf{Pelatihan}
    \begin{itemize}\kecil
      \item SGD momentum 0,9; lr 0,01; batch 40.
      \item Epoch: teks 100, citra 200, fusi 200.
      \item TensorFlow 1.12, Titan X.
    \end{itemize}
    \textbf{Metrik}
    \begin{itemize}\kecil
      \item OA, macro-F1, F1/kelas, \alert{oracle}, latensi.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Kontribusi, kelebihan, future work}
  \begin{columns}[T]
    \column{0.34\textwidth}
    \textbf{Kontribusi}
    \begin{itemize}\kecil
      \item Pipeline multimodal ringan dari komponen siap pakai (MobileNetV2 + FastText), dilatih end-to-end.
      \item Analisis derau OCR: $\sim$26\,\% kata di luar kamus.
      \item Bukti teks berderau tetap menaikkan akurasi.
      \item Rilis dataset teks \alert{QS-OCR}.
    \end{itemize}
    \column{0.32\textwidth}
    \textbf{Kelebihan}
    \begin{itemize}\kecil
      \item Desain sederhana, tiap pilihan diuji.
      \item Oracle dan F1 per kelas.
      \item Latensi dihitung termasuk OCR.
      \item Data teks terbuka sehingga mudah direplikasi.
    \end{itemize}
    \column{0.34\textwidth}
    \textbf{Future work}
    \begin{itemize}\kecil
      \item Data dunia nyata (foto, rotasi) + augmentasi.
      \item Segmentasi tata letak (dhSegment).
      \item Koreksi ejaan (SymSpell) + NER.
      \item Embedding tahan derau OCR.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Catatan kritis}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{itemize}\kecil
      \item Rerata 3 run tanpa simpangan baku dan uji signifikansi.
      \item Cara memilih 800 dokumen latih tidak dijelaskan.
      \item Hiperparameter disetel manual pada ``subset Tobacco3482''; tidak jelas terpisah dari data uji.
      \item OA teks 73,9 vs.\ 73,8 (Tabel 1 vs.\ 3).
    \end{itemize}
    \column{0.5\textwidth}
    \begin{itemize}\kecil
      \item RVL-CDIP hanya OA; tanpa F1 atau hasil per kelas.
      \item Baseline citra di bawah SOTA (89,1 vs.\ 90,97), sebagian karena tanpa \emph{multi-crop} saat inferensi (diakui penulis).
      \item \alert{Satu OCR saja}; tidak ada pembanding late fusion.
      \item FastText beku, jadi ``end-to-end'' hanya untuk CNN 1D.
    \end{itemize}
  \end{columns}
\end{frame}

\section{Paper 2 -- Kashyap et al.\ (2026)}

\begin{frame}{Latar belakang, dataset, modalitas}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Masalah}
    \begin{itemize}\kecil
      \item Unimodal tidak tangguh: visual ambigu, teks OCR penuh OOV.
      \item VLM besar (LayoutLMv3, DocFormer, DocLLM) terlalu berat untuk edge/IoMT.
      \item Fusi awal/menengah rentan propagasi galat antarmodal.
      \item Celah: trade-off akurasi--efisiensi jarang dibahas, sehingga diusulkan \alert{dua kerangka} terpisah.
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Dataset} (semua model dilatih di RVL-CDIP)\\[2pt]
    {\kecil\begin{tabular}{lrrc}
      \toprule Dataset & Total & Uji & Kelas \\ \midrule
      RVL-CDIP & 399.743 & 39.995 & 16 \\
      Tobacco3482 & 3.482 & 3.482 & 10 \\
      RealDoc-16 (primer) & 160 & 160 & 16 \\ \bottomrule
    \end{tabular}}
    \vspace{4pt}
    \textbf{Modalitas}
    \begin{itemize}\kecil
      \item Fusi: citra + teks OCR.
      \item SCSR: teks saja.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Pipeline}
  \centering
  \begin{minipage}{0.85\linewidth}\input{../gambar/pipeline-kashyap}\end{minipage}
  \begin{itemize}\kecil
    \item Pendekatan 1 mengejar akurasi: dua aliran dilatih terpisah, lalu logit dirata-rata.
    \item Pendekatan 2 mengejar efisiensi: pada dasarnya $k$-NN ($k=10$) di ruang embedding hasil pelatihan kontrastif.
  \end{itemize}
\end{frame}

\begin{frame}{Model unimodal, fusi, metrik}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Citra: EfficientNetB3}
    \begin{itemize}\kecil
      \item 300$\times$300, RGB, [0,1]; 1536 $\to$ Dropout $\to$ 16.
      \item 10 epoch, batch 16, Adam $10^{-3}$; $\approx$22,5 jam.
    \end{itemize}
    \textbf{Teks: BERT\textsubscript{base}}
    \begin{itemize}\kecil
      \item \awas{OCR komersial tanpa nama}; praproses: huruf kecil, buang angka/stopword/tanda baca.
      \item 10 epoch, batch 16, Adam $5\!\times\!10^{-5}$, dropout 0,3; $\approx$41 jam.
    \end{itemize}
    \textbf{Teks: SCSR (bge-small-en, 33,3 jt param.)}
    \begin{itemize}\kecil
      \item MNRL + Matryoshka; 4 epoch, AdamW $2\!\times\!10^{-5}$.
      \item Chunk 512 karakter, FAISS 1,03 jt $\times$ 384-d.
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Fusi: late fusion} \quad $\hat y = \arg\max_c \tfrac12 (z_v + z_t)_c$
    \begin{itemize}\kecil
      \item Rata-rata logit tak berbobot; tidak ada parameter fusi.
    \end{itemize}
    \textbf{Metrik}
    \begin{itemize}\kecil
      \item Acc; P/R/F1 \emph{macro} (RVL, RealDoc) dan \emph{weighted} (Tobacco).
      \item Akurasi per kelas; parameter, latensi, throughput.
      \item UMAP, Grad-CAM, integrated gradients.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Kontribusi, kelebihan, future work}
  \begin{columns}[T]
    \column{0.34\textwidth}
    \textbf{Kontribusi} (versi penulis)
    \begin{itemize}\kecil
      \item Late fusion dua aliran yang modular (EfficientNetB3 + BERT).
      \item SCSR: kontrastif terawasi + retrieval sebagai pengklasifikasi.
      \item Dataset primer RealDoc-16.
      \item Analisis Grad-CAM dan integrated gradients.
    \end{itemize}
    \column{0.33\textwidth}
    \textbf{Kelebihan}
    \begin{itemize}\kecil
      \item Jurnal 2026, open access (memenuhi syarat dosen).
      \item Trade-off akurasi--efisiensi eksplisit.
      \item Uji di tiga dataset, termasuk data primer.
      \item P/R/F1 + per kelas untuk tiap unimodal.
    \end{itemize}
    \column{0.33\textwidth}
    \textbf{Future work}
    \begin{itemize}\kecil
      \item Dokumen multibahasa dengan OCR yang sangat terdegradasi.
      \item PEFT untuk menekan memori pelatihan.
      \item Pembobotan modalitas adaptif / fusi berbasis keyakinan.
      \item Embedding citra ringan di indeks SCSR.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Catatan kritis (1/2): protokol dan perbandingan}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{itemize}\kecil
      \item \awas{OCR tidak disebut}, sehingga tidak bisa direplikasi.
      \item \awas{Pemetaan kelas Tobacco hilang}: \emph{note} + \emph{report} (13,4\,\%) tidak ada di RVL-CDIP, sehingga tanpa pemetaan akurasi maks.\ $\approx$86,6\,\%, padahal dilaporkan 92\,\%.
      \item Precision berbobot Tobacco (98,23) $\gg$ recall (92,00), pola sama di T dan V; konsisten dengan prediksi jatuh ke kelas di luar label Tobacco.
      \item Potensi kebocoran: Tobacco3482 dan RVL-CDIP dari arsip yang sama; tidak disebut ada deduplikasi.
    \end{itemize}
    \column{0.5\textwidth}
    \begin{itemize}\kecil
      \item \awas{Perbandingan selektif}: Kanchi [77] dan Bakkali [46] dikutip tetapi angkanya (95,7 dan 96,94 di Tobacco; Kanchi 95,48 di RVL) tidak dibandingkan. LayoutLMv3/DocFormer hanya disebut ``terlalu berat''.
      \item Protokol Tobacco berbeda dari Audebert (transfer vs.\ 800 latih).
      \item Praproses buang angka/stopword/tanda baca sebelum BERT (kritik yang sama dengan Kanchi).
      \item Hanya rata-rata tak berbobot; tanpa ablasi fusi, oracle, seed, atau uji signifikansi.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Catatan kritis (2/2): angka dan SCSR}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{itemize}\kecil
      \item \awas{Angka tidak konsisten}: macro-F1 SCSR 84 $>$ (P 84 + R 81)/2; macro-recall $\neq$ akurasi pada data seimbang.
      \item RealDoc-16: 160 vs.\ 120 dokumen; 72,73\,\% = 8/11 padahal 10/kelas.
      \item Angka per kelas Audebert di Tabel 5 \awas{tidak ditemukan} di paper asli.
      \item Tabel latensi: 555,27 s $\times$ 5,42 $\approx$ 3.010 dok.\ (fusi) vs.\ 547,19 s $\times$ 6,36 $\approx$ 3.480 dok.\ (SCSR), jadi jumlah dokumen uji berbeda dan total waktu hampir sama.
    \end{itemize}
    \column{0.5\textwidth}
    \begin{itemize}\kecil
      \item Klaim ``tanpa DL'' untuk SCSR padahal transformer 33 jt; indeks FAISS $\approx$1,6 GB.
      \item Syarat ``$>$512 token'', tetapi chunk 512 \emph{karakter} ($\sim$100--150 token).
      \item Dokumen uji di-embed utuh (Algoritma 6), sehingga dokumen panjang terpotong di 512 token.
      \item Voting top-10 per \emph{chunk}, bukan per dokumen, sehingga condong ke kelas yang teksnya panjang.
    \end{itemize}
  \end{columns}
\end{frame}

\section{Paper 3 -- Kanchi et al.\ (2022)}

\begin{frame}{Latar belakang, dataset, modalitas}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Masalah}
    \begin{itemize}\kecil
      \item Dokumen sering mirip secara visual tetapi berbeda isi (letter, memo, report).
      \item Model dua aliran yang ada bergantung pada \emph{pre-training} RVL-CDIP (320 rb citra) dan model besar.
      \item Pertanyaan: bisakah klasifikator multimodal \alert{efisien} tetap akurat dengan \alert{data sedikit} (Tobacco3482 saja)?
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Dataset}
    \begin{itemize}\kecil
      \item Tobacco3482: 10 kelas; partisi ala Afzal (20--100 sampel/kelas), latih dibagi 80/20.
      \item RVL-CDIP: hanya untuk eksperimen \emph{pre-training}.
      \item Dua skenario: ImageNet + RVL-CDIP, dan ImageNet saja (fokus).
    \end{itemize}
    \textbf{Modalitas}
    \begin{itemize}\kecil
      \item Citra + teks OCR; teks diproses hierarkis (kata $\to$ kalimat $\to$ dokumen).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Pipeline EmmDocClassifier}
  \input{../gambar/pipeline-kanchi}
  \vspace{6pt}
  \begin{itemize}\kecil
    \item Fitur citra dan teks berdimensi sama, dikonkatenasi ($2d$), lalu conv + pooling + FC.
    \item Rata-rata fitur juga dicoba, tetapi konkatenasi lebih baik.
  \end{itemize}
\end{frame}

\begin{frame}{Model unimodal, fusi, metrik}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Citra: EfficientNet-B0}
    \begin{itemize}\kecil
      \item 5,3 jt parameter ($\sim$5$\times$ lebih ringan dari ResNet-50).
      \item 384$\times$384, gray $\to$ RGB, bobot ImageNet.
      \item Adam $10^{-3}$, 20 epoch, lr turun bertahap ke $10^{-5}$.
    \end{itemize}
    \textbf{Teks: Tesseract 4 + BERT + HAN}
    \begin{itemize}\kecil
      \item Praproses: buang simbol, stopword, \emph{stemming}.
      \item BERT\textsubscript{base} fine-tune (4 lapis akhir), lalu dibekukan; embedding = jumlah 4 lapis akhir.
      \item HAN: BiGRU + atensi kata dan kalimat; 25 kalimat, 10 rb kata.
      \item RMSProp $10^{-3}$, batch 100, 20 epoch.
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Fusi: tingkat fitur}
    \[ X_{ens} = [X_{img}; X_{text}] \in \mathbb{R}^{2d} \]
    \begin{itemize}\kecil
      \item Adam $10^{-3}$, 20 epoch, batch 100.
      \item NVIDIA V100 32 GB.
    \end{itemize}
    \textbf{Metrik}
    \begin{itemize}\kecil
      \item Akurasi, \emph{relative error reduction}.
      \item Matriks konfusi, latensi, contoh benar/salah.
      \item \awas{Tidak ada F1/presisi/recall.}
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Hasil (Tobacco3482, akurasi \%)}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \kecil
    \begin{tabular}{lcccc}

      \item Kelas tersulit: \emph{scientific} (mirip report/adv., citra berderau).

\begin{frame}{Catatan kritis}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{itemize}\kecil
      \item \awas{Hasil unimodal teks dan ablasi FastText tanpa angka}, padahal aliran teks adalah klaim utama.
      \item \awas{Angka citra identik dengan Ferrando} (94,04 = 94,04; 86,0 $\approx$ 85,99), tanpa penjelasan.
      \item ``From scratch'' menyesatkan: memakai bobot ImageNet dan BERT pre-trained.
      \item Protokol partisi ambigu; tidak ada jumlah ulangan atau simpangan baku.
      \item BERT di-fine-tune pada ``korpus kami'', sehingga ada potensi kebocoran data uji.
      \item Stemming + buang stopword sebelum BERT merusak konteks.
    \end{itemize}
    \column{0.5\textwidth}
    \begin{itemize}\kecil
      \item HAN butuh batas kalimat, tetapi cara segmentasi teks OCR tidak dijelaskan.
      \item Klaim ``ukuran turun 80\,\%'' mengabaikan BERT (110 jt param.).
      \item Latensi 105 ms (V100) $\approx$ Audebert 110 ms (Titan X); klaim waktu latih tanpa angka.
      \item Evaluasi strategi embedding (Gbr.\ 5, ``F1 token sering muncul'') tidak dijelaskan.
      \item Hanya akurasi; RVL 95,48\,\% didukung matriks konfusi, tetapi tanpa P/R/F1 dan tanpa cara transfer 16 $\to$ 10 kelas.
      \item Tanpa kode/data.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Keterbatasan bersama}
  \begin{itemize}
    \item \textbf{Tidak ada modalitas tata letak.} Teks dipakai sebagai urutan kata tanpa posisi;
          keluarga LayoutLM ditolak karena berat, tetapi fitur tata letak ringan (kotak dari OCR)
          tidak diuji.
    \item \textbf{Kualitas dataset.} Larson et al.\ (EACL 2023): derau label RVL-CDIP $\sim$8,1\,\%
          (1,6--16,9\,\% per kelas) dan tumpang tindih latih--uji yang besar. Selisih $<$1 poin di
          RVL-CDIP sulit ditafsirkan, dan risiko kebocoran Kashyap makin kuat.
    \item \textbf{Kelas tersulit sama}: scientific/report, yang mirip secara visual dan isi, dengan
          citra berderau sehingga OCR gagal. Batas atasnya ditentukan OCR, bukan arsitektur.
    \item \textbf{Klaim efisiensi} tidak menghitung OCR (kecuali Audebert), dan latensi model
          ketiganya setara (105--184 ms).
    \item \textbf{Tanpa uji statistik} di ketiga paper.
  \end{itemize}
\end{frame}

\begin{frame}{Relevansi (1/2): keputusan desain Fase 2}
  \kecil
  \begin{itemize}
    \item \textbf{Protokol}: 800 latih terstratifikasi, sisanya uji, 3 split (seed 42/43/44), tanpa RVL-CDIP.
          Mengikuti jumlah latih Audebert dengan deviasi tertulis (split acak, bukan k-fold).
          Pembanding sepadan: Audebert dan Kanchi-tanpa-RVL (dengan catatan protokol); angka Kashyap bukan target.
    \item \textbf{Faktor OCR}: QS-OCR (Tesseract 4, TIF; acuan Fase 1), Tesseract 5 (JPG), EasyOCR 1.7.2,
          PaddleOCR 3.7.0; Keras-OCR opsional. \alert{Daftar mesin perlu konfirmasi dosen.}
    \item \textbf{Faktor embedding}: FastText + CNN 1D (Fase 1) vs.\ BERT\textsubscript{base} fine-tune (512 token);
          BERT + HAN (Kanchi) sebagai kandidat level ketiga.
    \item \textbf{Praproses}: tanpa buang angka/stopword/tanda baca, supaya efek embedding tidak tercampur.
    \item \textbf{Fusi}: konkatenasi vs.\ rata-rata logit pada kondisi sama, + oracle + uji missing modality.
    \item \textbf{Evaluasi}: OA, macro-F1, F1/kelas, \alert{akurasi tiap unimodal}, rerata $\pm$ std 3 seed, McNemar.
    \item \textbf{Tata letak}: kotak posisi teks dari OCR disimpan untuk tahap lanjutan.
  \end{itemize}
\end{frame}

\begin{frame}{Relevansi (2/2): hipotesis kerja}
  Titik awal Fase 1 (eksekusi nyata): cabang teks Tesseract + FastText + CNN 1D mencapai
  OA $0{,}7266 \pm 0{,}0059$, tetapi sumbangannya di fusi kecil.

  \vspace{6pt}
  \begin{description}
    \item[H1] Mesin OCR mengubah akurasi cabang teks secara nyata (McNemar antar-mesin).
    \item[H2] BERT memperkecil selisih antar-OCR dibanding FastText (interaksi OCR $\times$ embedding).
    \item[H3] Cabang teks yang lebih kuat membuat fusi lebih \emph{memakai} teks (penurunan lebih
          besar saat teks dihilangkan); rata-rata logit lebih tahan terhadap cabang yang timpang
          daripada konkatenasi.
  \end{description}
  \vspace{4pt}
  {\kecil Status: belum-dijalankan.}
\end{frame}

\begin{frame}{Kesimpulan}
  \begin{itemize}
    \item \textbf{Audebert}: baseline sederhana dan transparan; kelemahannya satu OCR dan tanpa variansi.
    \item \textbf{Kanchi}: 90,3\,\% tanpa RVL-CDIP lewat BERT + HAN, tetapi hasil teks tidak
          dilaporkan, protokol ambigu, dan klaim efisiensi mengabaikan BERT.
    \item \textbf{Kashyap}: late fusion sederhana dengan akurasi tinggi dan alternatif retrieval
          ringan, tetapi pelaporannya bermasalah (OCR anonim, pemetaan kelas, angka tidak
          konsisten, perbandingan selektif).
    \item Ketiganya tidak mengisolasi OCR, tidak memakai tata letak, dan tidak menguji signifikansi.
          Celah inilah yang diisi Fase 2.
  \end{itemize}
\end{frame}
